Class Pet
    Attributes: name, age
    Constructor(name, age)
        this.name = name
        this.age = age
    Method displayInfo()
        print name, age

Class Dog extends Pet
    Attribute: breed
    Constructor(name, age, breed)
        call super(name, age)
        this.breed = breed
    Method displayInfo()
        print "Dog:", name, age, breed

Class Cat extends Pet
    Attribute: color
    Constructor(name, age, color)
        call super(name, age)
        this.color = color
    Method displayInfo()
        print "Cat:", name, age, color

Class Bird extends Pet
    Attribute: canFly
    Constructor(name, age, canFly)
        call super(name, age)
        this.canFly = canFly
    Method displayInfo()
        print "Bird:", name, age, canFly

Main
    dog = Dog("Bruno", 3, "Labrador")
    cat = Cat("Milo", 2, "White")
    bird = Bird("Parrot", 1, true)

    dog.displayInfo()
    cat.displayInfo()
    bird.displayInfo()
