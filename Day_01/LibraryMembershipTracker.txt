Class Member
    Attributes: name, ID, membershipDate, expiryDate
    Constructor(name, ID, membershipDate, expiryDate)
        set values
    Method isMembershipValid(currentDate)
        if currentDate <= expiryDate
            return true
        else
            return false

Main
    member = Member("Alice", 101, "2025-01-01", "2025-12-31")
    if member.isMembershipValid("2025-08-19")
        print "Membership Active"
    else
        print "Membership Expired"
