Class CallCenter
    Attribute: queue normalQueue, queue emergencyQueue
    Method addCustomer(name, type)
        if type == "emergency"
            enqueue emergencyQueue(name)
        else
            enqueue normalQueue(name)

    Method serveCustomer()
        if emergencyQueue not empty
            print "Serving:", dequeue emergencyQueue
        else if normalQueue not empty
            print "Serving:", dequeue normalQueue
        else
            print "No customers"

Main
    center = CallCenter()
    center.addCustomer("John", "normal")
    center.addCustomer("Mike", "emergency")
    center.addCustomer("Lucy", "normal")

    center.serveCustomer() 
    center.serveCustomer()
