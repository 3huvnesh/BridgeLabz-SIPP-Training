// QuizTimer.java
CLASS QuizTimer {
    INT[] timer
    INT currentIndex
    INT totalTime

    QuizTimer(INT totalQuestions, INT timePerQuestion) {
        this.totalTime = totalQuestions * timePerQuestion
        this.timer = NEW INT[this.totalTime]
        this.currentIndex = 0
        FOR i FROM 0 TO this.totalTime - 1
            this.timer[i] = i + 1 // Time in seconds
        END FOR
    }

    // Method to simulate time passing
    METHOD tick() {
        this.currentIndex = (this.currentIndex + 1) % this.totalTime
        IF this.timer[this.currentIndex] >= (this.totalTime * 0.8) THEN
            PRINT "WARNING: Time is running out!"
        END IF
    }

    // Method to get remaining time
    METHOD getRemainingTime() RETURNS INT {
        RETURN this.totalTime - this.timer[this.currentIndex]
    }
}

// Main.java (Example Usage)
MAIN {
    QuizTimer quiz = NEW QuizTimer(5, 60) // 5 questions, 60s each
    FOR i FROM 1 TO 250 // Simulate 250 seconds
        quiz.tick()
    END FOR
    
    PRINT "Remaining time: " + quiz.getRemainingTime() // Will show a warning
}
