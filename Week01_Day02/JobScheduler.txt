// Job.java (Data class)
CLASS Job {
    STRING name
    INT deadline
    INT processingTime

    Job(STRING name, INT deadline, INT processingTime) {
        this.name = name
        this.deadline = deadline
        this.processingTime = processingTime
    }
}

// JobScheduler.java
CLASS JobScheduler {
    // PriorityQueue to manage jobs
    // The comparator prioritizes earlier deadlines, then shorter processing times
    PriorityQueue<Job> jobQueue = NEW PriorityQueue<Job>((job1, job2) -> {
        IF job1.deadline != job2.deadline THEN
            RETURN job1.deadline - job2.deadline
        ELSE
            RETURN job1.processingTime - job2.processingTime
        END IF
    })

    METHOD addJob(Job job) {
        this.jobQueue.add(job)
    }

    METHOD scheduleNextJob() {
        IF this.jobQueue.isEmpty() THEN
            PRINT "No jobs to schedule."
            RETURN
        END IF
        
        Job nextJob = this.jobQueue.poll()
        PRINT "Scheduling job: " + nextJob.name + " (Deadline: " + nextJob.deadline + ", Time: " + nextJob.processingTime + ")"
    }
}

// Main.java (Example Usage)
MAIN {
    JobScheduler scheduler = NEW JobScheduler()
    scheduler.addJob(NEW Job("Task A", 5, 2))
    scheduler.addJob(NEW Job("Task B", 3, 1))
    scheduler.addJob(NEW Job("Task C", 5, 1))

    scheduler.scheduleNextJob() // Task B (deadline 3)
    scheduler.scheduleNextJob() // Task C (deadline 5, time 1)
    scheduler.scheduleNextJob() // Task A (deadline 5, time 2)
}
