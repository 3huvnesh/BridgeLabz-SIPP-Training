
Interface Searchable
    Method search(keyword)

Interface Printable
    Method print()

Class LabReport implements Searchable, Printable
    Attribute: reportDetails
    Constructor(details)
    Method search(keyword)
        if keyword in reportDetails
            return "Found in LabReport"
        else return "Not found"
    Method print()
        print "Lab Report:", reportDetails

Class Prescription implements Searchable, Printable
    Attribute: medicineList
    Constructor(list)
    Method search(keyword)
        if keyword in medicineList
            return "Medicine Found"
        else return "Not found"
    Method print()
        print "Prescription:", medicineList

Main
    report = LabReport("Blood test: Normal")
    prescription = Prescription(["Paracetamol", "Vitamin C"])

    report.print()
    prescription.print()

    print report.search("Blood")
    print prescription.search("Vitamin C")
