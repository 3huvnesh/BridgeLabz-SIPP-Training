// Flight.java (Data class)
CLASS Flight {
    STRING flightNumber
    FLOAT price
    INT durationInHours
}

// FlightComparator.java
CLASS FlightComparator IMPLEMENTS Comparator<Flight> {
    // Compares based on price, then duration
    OVERRIDE METHOD compare(Flight f1, Flight f2) RETURNS INT {
        IF f1.price != f2.price THEN
            RETURN f1.price - f2.price // Sort by price (ascending)
        ELSE
            RETURN f1.durationInHours - f2.durationInHours // Sort by duration (ascending)
        END IF
    }
}

MAIN {
    LIST<Flight> flights = NEW ArrayList<Flight>()
