// Item.java (Data class)
CLASS Item {
    STRING name
    FLOAT price

    Item(STRING name, FLOAT price) {
        this.name = name
        this.price = price
    }
    
    METHOD getPrice() RETURNS FLOAT {
        RETURN this.price
    }
}

// ShoppingCart.java
CLASS ShoppingCart {
    LIST<Item> cart = NEW ArrayList<Item>()

    METHOD addItem(Item item) {
        this.cart.add(item)
    }

    // Generic method to calculate total price
    GENERIC METHOD <T EXTENDS Item> calculateTotalPrice(LIST<T> items) RETURNS FLOAT {
        FLOAT total = 0.0
        FOR item IN items
            total += item.getPrice()
        END FOR
        RETURN total
    }
    
    // Method to call the generic method on the cart
    METHOD getCartTotal() RETURNS FLOAT {
        RETURN calculateTotalPrice(this.cart)
    }
}

// Main.java (Example Usage)
MAIN {
    ShoppingCart myCart = NEW ShoppingCart()
    myCart.addItem(NEW Item("Laptop", 1200.00))
    myCart.addItem(NEW Item("Mouse", 25.50))
    
    FLOAT total = myCart.getCartTotal()
    PRINT "Total cart price: $" + total
}
