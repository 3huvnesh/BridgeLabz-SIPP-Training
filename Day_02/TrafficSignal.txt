// TrafficLight.java (Enum)
ENUM TrafficLight {
    RED,
    YELLOW,
    GREEN
}

// TrafficSignal.java
CLASS TrafficSignal {
    TrafficLight currentState

    // Constructor to initialize the state
    TrafficSignal() {
        this.currentState = TrafficLight.RED
    }

    // Method to change the light
    METHOD changeLight() {
        SWITCH this.currentState
            CASE TrafficLight.RED:
                this.currentState = TrafficLight.GREEN
                BREAK
            CASE TrafficLight.GREEN:
                this.currentState = TrafficLight.YELLOW
                BREAK
            CASE TrafficLight.YELLOW:
                this.currentState = TrafficLight.RED
                BREAK
        END SWITCH
    }

    // Method to get the current state
    METHOD getCurrentState() RETURNS TrafficLight {
        RETURN this.currentState
    }
}

// Main.java (Example Usage)
MAIN {
    TrafficSignal signal = NEW TrafficSignal()
    PRINT "Initial state: " + signal.getCurrentState() // Prints RED
    
    signal.changeLight()
    PRINT "Current state: " + signal.getCurrentState() // Prints GREEN

    signal.changeLight()
    PRINT "Current state: " + signal.getCurrentState() // Prints YELLOW
}
