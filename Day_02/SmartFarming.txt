// Crop.java (Abstract Class)
ABSTRACT CLASS Crop {
    STRING name
    FLOAT yield

    // Abstract methods to be implemented by subclasses
    ABSTRACT METHOD waterRequirement() RETURNS STRING
    ABSTRACT METHOD harvest() RETURNS STRING
    
    // Concrete method
    METHOD getInfo() {
        PRINT "Crop: " + this.name + ", Yield: " + this.yield
    }
}

// Rice.java
CLASS Rice EXTENDS Crop {
    Rice(FLOAT yield) {
        this.name = "Rice"
        this.yield = yield
    }

    OVERRIDE METHOD waterRequirement() RETURNS STRING {
        RETURN "High water requirement."
    }

    OVERRIDE METHOD harvest() RETURNS STRING {
        RETURN "Harvesting rice by cutting stalks and threshing."
    }
}

// Wheat.java
CLASS Wheat EXTENDS Crop {
    Wheat(FLOAT yield) {
        this.name = "Wheat"
        this.yield = yield
    }

    OVERRIDE METHOD waterRequirement() RETURNS STRING {
        RETURN "Moderate water requirement."
    }

    OVERRIDE METHOD harvest() RETURNS STRING {
        RETURN "Harvesting wheat with a combine harvester."
    }
}

// Main.java (Example Usage)
MAIN {
    Crop riceField = NEW Rice(150.5)
    Crop wheatField = NEW Wheat(120.0)

    riceField.getInfo()
    PRINT riceField.waterRequirement()
    PRINT riceField.harvest()
    
    PRINT "---"
    
    wheatField.getInfo()
    PRINT wheatField.waterRequirement()
    PRINT wheatField.harvest()
}
